document.addEventListener('DOMContentLoaded', function () {
  const provinceSelect = document.getElementById('perm_province');
  const districtSelect = document.getElementById('perm_district');
  const municipalitySelect = document.getElementById('perm_municipality');
  const wardSelect = document.getElementById('perm_ward');

  // Fetch provinces on load
  fetch('frontend/maker_a/get_address_data.php?type=province')
    .then(res => res.json())
    .then(data => {
      provinceSelect.innerHTML = `<option value="">Select Province</option>`;
      data.forEach(p => {
        provinceSelect.innerHTML += `<option value="${p}">${p}</option>`;
      });
    });

  // On province change → load districts
  provinceSelect.addEventListener('change', function () {
    const province = this.value;
    districtSelect.innerHTML = `<option>Loading...</option>`;
    fetch(`get_address_data.php?type=district&province=${province}`)
      .then(res => res.json())
      .then(data => {
        districtSelect.innerHTML = `<option value="">Select District</option>`;
        data.forEach(d => {
          districtSelect.innerHTML += `<option value="${d}">${d}</option>`;
        });
      });
  });

  // On district change → load municipalities
  districtSelect.addEventListener('change', function () {
    const district = this.value;
    municipalitySelect.innerHTML = `<option>Loading...</option>`;
    fetch(`get_address_data.php?type=municipality&district=${district}`)
      .then(res => res.json())
      .then(data => {
        municipalitySelect.innerHTML = `<option value="">Select Municipality</option>`;
        data.forEach(m => {
          municipalitySelect.innerHTML += `<option value="${m}">${m}</option>`;
        });
      });
  });

  // On municipality change → load ward count
  municipalitySelect.addEventListener('change', function () {
    const municipality = this.value;
    wardSelect.innerHTML = `<option>Loading...</option>`;
    fetch(`get_address_data.php?type=ward&municipality=${municipality}`)
      .then(res => res.json())
      .then(data => {
        wardSelect.innerHTML = `<option value="">Select Ward</option>`;
        data.forEach(w => {
          wardSelect.innerHTML += `<option value="${w}">${w}</option>`;
        });
      });
  });
});
